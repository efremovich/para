Замер времени прохождение документа за промежуток времени.
```
WITH selectedDocs AS (
    SELECT d.id 
    FROM docs d
    WHERE d.created::timestamptz > '2026-04-23 18:00:00'
      AND d.created::timestamptz < '2026-04-23 18:00:10'
)
SELECT 
    AVG(diff) AS avg_processing_time
FROM (
    SELECT 
        l.inbox,
        l.created,
        l.processed,
        CASE 
            WHEN l.inbox = false THEN
                l.processed::timestamptz 
                - LAG(l.created::timestamptz) OVER (ORDER BY l.created::timestamptz)
        END AS diff
    FROM ls2docs ld
    JOIN ls l ON l.id = ld.ls
    WHERE ld.doc IN (SELECT id FROM selectedDocs)
) t
WHERE t.inbox = false
  AND diff IS NOT NULL;
```

Расчет ведется от даты создания входящей ЛС под дату окончания процессинга у исходящей ЛС.
