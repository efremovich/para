```sql
select -- Год к Году по Дням  
toDayOfYear(day) as dy,  
toYear(day) as y,  
"day",  
all_docs,  
prim  
from (  
select  
toStartOfDay(create_date) as day,  
count(*) as all_docs,  
countIf(d.primary_doc) as prim  
from docs d  
where create_date between '2022-01-01' and now()  
group by day  
order by day  
) dd;
```
